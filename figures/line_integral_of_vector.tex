\documentclass{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usetikzlibrary{arrows.meta}

\begin{document}

\definecolor{tab_blue}{HTML}{1F77B4}
\definecolor{tab_orange}{HTML}{FF7F0E}
\definecolor{tab_green}{HTML}{2CA02C}
\definecolor{tab_red}{HTML}{D62728}

\begin{tikzpicture}
\begin{axis}[
% axis on top,
% axis lines = box,
axis lines = none,
unit vector ratio=1 1, 
% axis equal=false,
scale=2.0, 
xmin=-1.6, xmax=2.6, ymin=-2.0, ymax=2.8, 
grid = major,
declare function={ 
    myfunc(\x) = 0.5*\x+0.3*sin(deg(pi*\x));
    myfuncderiv(\x) = 0.5 + 0.3*pi*cos(deg(pi*\x)); 
    vecx(\x) = 0.15 + 0.10 * \x; 
    vecy(\x) = 0.3 - 0.1 * sqrt(\x);
}
]

\pgfmathsetmacro{\a}{0.0}
\pgfmathsetmacro{\b}{2.0}
\pgfmathtruncatemacro{\Nsub}{8}
\pgfmathsetmacro{\h}{(\b-\a)/\Nsub}

\addplot[domain=\a:\b, samples=100, ultra thick] {myfunc(x)};

\pgfplotsforeachungrouped \i in {1,...,\Nsub-1}{
    \pgfmathsetmacro{\xcoord}{\i*\h} 
    \pgfmathsetmacro{\ycoord}{myfunc(\xcoord)} 
    \pgfmathsetmacro{\temp}{vecx(\xcoord)*1.0/(sqrt(1.0+myfuncderiv(\xcoord)^2)) + vecy(\xcoord)*myfuncderiv(\xcoord)/(sqrt(1.0+myfuncderiv(\xcoord)^2))}
    \edef\doplot{\noexpand 
        \node at (\xcoord,\ycoord) {$\bullet$};
    }
    \doplot
    \edef\doplot{\noexpand 
        \draw[->, >=stealth, ultra thick, color=tab_red] (\xcoord,\ycoord) -- ({\xcoord+\temp*1.0/(sqrt(1.0+myfuncderiv(\xcoord)^2))}, {\ycoord+\temp*myfuncderiv(\xcoord)/(sqrt(1.0+myfuncderiv(\xcoord)^2))});
    }
    \doplot
    \edef\doplot{\noexpand 
        \draw[->, >=stealth, thick] (\xcoord,\ycoord) -- ({\xcoord+vecx(\xcoord)}, {\ycoord+vecy(\xcoord)});
    }
    \doplot
}

\end{axis}
\end{tikzpicture}

\end{document}
