\documentclass{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{document}

\definecolor{tab_blue}{HTML}{1F77B4}
\definecolor{tab_orange}{HTML}{FF7F0E}
\definecolor{tab_green}{HTML}{2CA02C}
\definecolor{tab_red}{HTML}{D62728}

\begin{tikzpicture}
\begin{axis}[
% axis on top,
axis lines = middle,
unit vector ratio=1 1, 
% axis equal=false,
scale=1.2, 
xmin=-1.8, xmax=3.4, ymin=-1.1, ymax=1.10, 
grid = major,
xtick={-pi/2, -pi/4, 0, pi/4, pi/2, 3*pi/4, pi},
ytick={-1, -1/2, 0, 1/2, 1},
xticklabels = {$-\pi/2$,,$0$,, $\pi/2$,, $\pi$},
yticklabels = {$-1$,, $0$,, $1$},
xlabel=$x$,
ylabel=$y$,
xlabel style={below right},
ylabel style={above left},
axis line style = {->},
]

\addplot[domain=-pi/2:pi/2, color=tab_red, very thick] {sin(deg(x))};
\addplot[domain=0:pi, color=tab_blue, very thick] {cos(deg(x))};
\legend{$y=\sin x$, $y=\cos x$}
\end{axis}
\end{tikzpicture}

\end{document}
