\documentclass{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{document}

\definecolor{tab_blue}{HTML}{1F77B4}
\definecolor{tab_orange}{HTML}{FF7F0E}
\definecolor{tab_green}{HTML}{2CA02C}

\begin{tikzpicture}
\begin{axis}[
% axis on top,
axis lines = middle,
unit vector ratio=1 1, 
% axis equal=false,
scale=1.2, 
xmin=-1.8, xmax=1.8, ymin=-3.8, ymax=3.8, 
grid = major,
xtick={-pi/2, -pi/4, 0, pi/4, pi/2},
ytick={-3, -2, -1, 0, 1, 2, 3},
xticklabels = {$-\pi/2$,,$0$,, $\pi/2$},
yticklabels = {,,$-1$, $0$, $1$,,},
xlabel=$x$,
ylabel=$y$,
xlabel style={below right},
ylabel style={above left},
axis line style = {->},
]

\addplot[samples=100, domain=-pi/2+0.1:pi/2-0.1, color=tab_green, very thick] {tan(deg(x))};
\legend{$y=\tan x$}
\end{axis}
\end{tikzpicture}
